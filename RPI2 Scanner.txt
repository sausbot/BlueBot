def calculateDist(rssi):
#Calculates the distance where -54 is the power 1 m away
#2.5 is an environmental factor
d = 10 ** ((-54 -  rssi)/(10*2.5))
d = round (d,2)
return d

#Kalman Filter to smooth RSSI values in a rolling window array
def updateArray (rssi_array, new_rssi):
        if len(rssi_array) == max_samples:
                del rssi_array[0]
        if len(rssi_array) < max_samples:
                rssi_array.append (new_rssi)

        #print ("DISTANCE WINDOW {}".format(rssi_array))
